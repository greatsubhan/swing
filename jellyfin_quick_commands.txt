============================================================
 JELLYFIN ON ANDROID - Quick Setup (copy/paste to Termux)
============================================================

Run these lines ONE AT A TIME in Termux on the phone:

--- Part 1: Install Ubuntu ---

pkg update -y && pkg upgrade -y
pkg install proot-distro -y
proot-distro install ubuntu
proot-distro login ubuntu

--- Part 2: Install Jellyfin (run INSIDE ubuntu) ---

# First, stop whatever is using port 8080:
fuser -k 8080/tcp 2>/dev/null || true

apt update -y && apt upgrade -y
apt install -y curl gnupg apt-transport-https ca-certificates
curl -fsSL https://repo.jellyfin.org/jellyfin_team.gpg.key | gpg --dearmor -o /etc/apt/trusted.gpg.d/jellyfin.gpg
echo "deb [arch=$(dpkg --print-architecture)] https://repo.jellyfin.org/ubuntu jammy main" > /etc/apt/sources.list.d/jellyfin.list
apt update -y
apt install -y jellyfin

# Create start script with port 8080
cat > /opt/start_jellyfin.sh << 'EOF'
#!/bin/bash
export JELLYFIN_DATA_DIR="/var/lib/jellyfin"
export JELLYFIN_CONFIG_DIR="/etc/jellyfin"
export JELLYFIN_CACHE_DIR="/var/cache/jellyfin"
export JELLYFIN_LOG_DIR="/var/log/jellyfin"
mkdir -p $JELLYFIN_DATA_DIR $JELLYFIN_CONFIG_DIR $JELLYFIN_CACHE_DIR $JELLYFIN_LOG_DIR
nohup /usr/bin/jellyfin --datadir "$JELLYFIN_DATA_DIR" --configdir "$JELLYFIN_CONFIG_DIR" --cachedir "$JELLYFIN_CACHE_DIR" --logdir "$JELLYFIN_LOG_DIR" --port 8080 > /var/log/jellyfin/console.log 2>&1 &
echo "Jellyfin started on port 8080"
EOF

chmod +x /opt/start_jellyfin.sh

# Start Jellyfin
bash /opt/start_jellyfin.sh

# Exit Ubuntu
exit

--- Part 3: Media Folders (in Termux) ---

proot-distro login ubuntu
mkdir -p /media/movies /media/tvshows /media/music
# Exit and copy media from phone storage to Termux home:
# In Termux (OUTSIDE ubuntu):
# cp -r /sdcard/Movies ~/media/
# cp -r /sdcard/Music ~/media/
# Then inside ubuntu symlink them:
# ln -s /root/media/movies /media/movies

--- Part 4: Auto-start on boot ---

# Install termux-boot from F-Droid first
# Then in Termux run:
pkg install termux-services -y
mkdir -p ~/.termux/boot/
cat > ~/.termux/boot/start_jellyfin.sh << 'EOF'
#!/data/data/com.termux/files/usr/bin/bash
termux-wake-lock
# Kill anything on port 8080 first, then start jellyfin
fuser -k 8080/tcp 2>/dev/null || true
proot-distro login ubuntu -- bash /opt/start_jellyfin.sh
EOF
chmod +x ~/.termux/boot/start_jellyfin.sh
